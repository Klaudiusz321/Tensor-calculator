# ===========================================
# Przykładowa metryka: Metryka Schwarzschilda
# Opisuje czasoprzestrzeń wokół nierorotującej czarnej dziury
# ===========================================

# Format: współrzędne ; parametry
t, r, theta, phi ; M

# Elementy tensora metrycznego g_ij
# Format: indeks_i indeks_j wyrażenie

# g_00 = -(1 - 2M/r) - składowa czasowa
0 0 -(1 - 2*M/r)

# g_11 = 1/(1 - 2M/r) - składowa radialna
1 1 1/(1 - 2*M/r)

# g_22 = r^2 - składowa theta
2 2 r**2

# g_33 = r^2 * sin^2(theta) - składowa phi
3 3 r**2 * sin(theta)**2
